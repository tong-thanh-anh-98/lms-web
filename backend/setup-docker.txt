- Instructions for installing and configuring docker-compose for the project.
- Laravel + React + Docker Setup:

- Cấu trúc thư mục:
.
├── backend/              # Laravel app
├── frontend/             # ReactJS + Vite
├── docker/
│   ├── php/Dockerfile    # PHP-FPM config
│   ├── nginx/default.conf # Nginx config
│   └── docker-compose.yml
└── README.md

- Yêu cầu:
    + Docker
    + Docker Compose
    + WSL2 (Windows/Linux), macOS, hoặc Linux

- Cấu hình .env (Laravel):
APP_NAME=Laravel
APP_ENV=local
APP_KEY=base64:...
APP_DEBUG=true
APP_URL=http://localhost

DB_CONNECTION=mysql
DB_HOST=mysql
DB_PORT=3306
DB_DATABASE=db_lms
DB_USERNAME=root
DB_PASSWORD=root

=> chú ý: DB_HOST=mysql phải trùng tên service trong docker-compose, KHÔNG để 127.0.0.1.

- Cấu hình Docker: docker-compose.yml
version: '3.8'

services:
  # Laravel (PHP-FPM)
  app:
    build:
      context: ../
      dockerfile: ./docker/php/Dockerfile
    container_name: laravel-app
    working_dir: /var/www/html
    volumes:
      - ../backend:/var/www/html
    depends_on:
      - mysql
    environment:
      - DB_HOST=mysql
      - DB_PORT=3306
    # restart: always
    networks:
      - app-network

  # Nginx
  web:
    image: nginx:alpine
    container_name: nginx
    ports:
      - "80:80"
    volumes:
      - ../backend:/var/www/html
      - ./nginx/default.conf:/etc/nginx/conf.d/default.conf
    depends_on:
      - app
    # restart: always
    networks:
      - app-network

  # MySQL
  mysql:
    image: mysql:8.0
    container_name: mysql
    ports:
      - "3306:3306"
    environment:
      MYSQL_ROOT_PASSWORD: root
      MYSQL_DATABASE: db_lms
    volumes:
      - mysql-data:/var/lib/mysql
    # restart: always
    networks:
      - app-network

  # ReactJS + Vite
  frontend:
    image: node:23.6.0
    container_name: react-app
    working_dir: /app
    volumes:
      - ../frontend:/app
      - /app/node_modules  #  mount riêng node_modules nội bộ
    ports:
      - "5173:5173"
    command: sh -c "npm install && npm run dev"
    # restart: always  # hoặc bạn có thể xóa nếu không muốn auto restart trong dev
    networks:
      - app-network

  # phpMyAdmin
  phpmyadmin:
    image: phpmyadmin/phpmyadmin
    container_name: phpmyadmin
    ports:
      - "8080:80"
    environment:
      PMA_HOST: mysql
      PMA_PORT: 3306
      PMA_USER: root
      PMA_PASSWORD: root
    depends_on:
      - mysql
    # restart: always
    networks:
      - app-network

volumes:
  mysql-data:

networks:
  app-network:
    driver: bridge

- default.conf (nginx):
server {
    listen 80;
    server_name localhost;
    root /var/www/html/public;
    index index.php index.html index.htm;

    location / {
        try_files $uri $uri/ /index.php?$query_string;
    }

    location ~ \.php$ {
        fastcgi_pass app:9000;
        fastcgi_index index.php;
        include fastcgi.conf;
    }

    location ~ /\.(?!well-known).* {
        deny all;
    }

    location ~* \.(env|bak|config|sql)$ {
        deny all;
    }

    error_log  /var/log/nginx/error.log;
    access_log /var/log/nginx/access.log;
}


Cách chạy dự án:
1. Build và khởi chạy toàn bộ dịch vụ: docker-compose up -d --build

2. Truy cập:
    Laravel Backend: http://localhost

    ReactJS Frontend (Vite): http://localhost:5173

    phpMyAdmin: http://localhost:8080

    User: root

    Password: root

3. Khởi tạo Laravel project (nếu chưa có):
    docker exec -it laravel-app composer install
    docker exec -it laravel-app php artisan migrate
    docker exec -it laravel-app php artisan key:generate

4. Lệnh tiện ích:
    # Xem log Laravel
    docker logs laravel-app

    # Truy cập bash trong container Laravel
    docker exec -it laravel-app bash

    # Dừng toàn bộ container
    docker-compose down

5. Ghi chú:
    Laravel root: ./backend

    React root: ./frontend

    Port 80, 5173, 8080 nên không trùng với dự án khác

    Nếu dùng nhiều dự án Docker, nên đổi port/nginx/ngăn trùng container_name
